**Problem Statement**

The **Fibonacci numbers**, commonly denoted `F(n)` form a sequence, called the **Fibonacci sequence**, such that each number is the sum of the two preceding ones, starting from `0` and `1`. That is,

```
F(0) = 0, F(1) = 1
F(n) = F(n - 1) + F(n - 2), for n > 1.
```

Given `n`, calculate `F(n)`.

**Example 1:**

```
Input: n = 2
Output: 1
Explanation: F(2) = F(1) + F(0) = 1 + 0 = 1.
```

**Example 2:**

```
Input: n = 3
Output: 2
Explanation: F(3) = F(2) + F(1) = 1 + 1 = 2.
```

**Example 3:**

```
Input: n = 4
Output: 3
Explanation: F(4) = F(3) + F(2) = 2 + 1 = 3.
```

**Constraints:**

- `0 <= n <= 30`

**Approach 1: Plain Recursion (Brute Force)**

```cpp
class Solution {
public:
    int fib(int n) {
        if (n <= 1) return n;
        return fib(n - 1) + fib(n - 2);
    }
};
```

- Time Complexity: O(2^N)
- Space Complexity: O(N) - Recursion stack depth

**Approach 2: Memoization (Top-Down Dynamic Programming)**

Store computed values in a memoization table/vector so each subproblem is solved only once

```cpp
class Solution {
private:
    int solve(int n, vector<int>& dp) {
        if (n <= 1) return n;
        if (dp[n] != -1) return dp[n];

        return dp[n] = solve(n - 1, dp) + solve(n - 2, dp);
    }

public:
    int fib(int n) {
        vector<int> dp(n + 1, -1);
        return solve(n, dp);
    }
};
```

- Time Complexity: O(N) - Computes states 0 to N once.
- Space Complexity: O(N) - dp vector + Recursion stack depth

Approach 3: Iterative DP

```cpp
class Solution {
public:
    int fib(int n) {
        if (n <= 1) return n;
        vector<int> dp(n + 1, 0);
        dp[1] = 1;
        for (int i = 2; i <= n; i++) {
            dp[i] = dp[i - 1] + dp[i - 2];
        }

        return dp[n];
    }
};
```

- Time Complexity: O(N) - Computes states 0 to N once.
- Space Complexity: O(N) - dp vector

Approach 4: Space-Optimized Iterative DP (Standard Optimal)

```cpp
class Solution {
public:
    int fib(int n) {
        if (n <= 1) return n;

        int prev2 = 0; // F(0)
        int prev1 = 1; // F(1)

        for (int i = 2; i <= n; i++) {
            int curr = prev1 + prev2;
            prev2 = prev1;
            prev1 = curr;
        }

        return prev1;
    }
};
```

- Time Complexity: O(N) - Single loop from 2 to N
- Space Complexity: O(1)

Approach 5: Matrix Exponentiation (OA / Large Constraints Optimal)

Express the linear recurrence relation using matrix multiplication:

Using **Binary Exponentiation**, compute M^(n-1) in O(logN) time.

```cpp
class Solution {
private:
    vector<vector<long long>> multiply(vector<vector<long long>>& A, 
                                                vector<vector<long long>>& B) {
        vector<vector<long long>> C(2, vector<long long>(2, 0));
        C[0][0] = A[0][0] * B[0][0] + A[0][1] * B[1][0];
        C[0][1] = A[0][0] * B[0][1] + A[0][1] * B[1][1];
        C[1][0] = A[1][0] * B[0][0] + A[1][1] * B[1][0];
        C[1][1] = A[1][0] * B[0][1] + A[1][1] * B[1][1];
        return C;
    }

    vector<vector<long long>> matrixPower(vector<vector<long long>> A, int p) {
        vector<vector<long long>> res = {{1, 0}, {0, 1}}; // Identity matrix
        while (p > 0) {
            if (p & 1) res = multiply(res, A);
            A = multiply(A, A);
            p >>= 1;
        }
        return res;
    }

public:
    int fib(int n) {
        if (n <= 1) return n;

        vector<vector<long long>> A = {{1, 1}, {1, 0}};
        A = matrixPower(A, n - 1);

        return A[0][0]; // A[0][0]*F(1) + A[0][1]*F(0) = A[0][0]*1 + 0 = A[0][0]
    }
};

```

- Time Complexity: O(logN) - Binary exponentiation of a 2X2 matrix.
- Space Complexity: O(1)
